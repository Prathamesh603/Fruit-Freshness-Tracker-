cmake_minimum_required(VERSION 3.22.1)
project(fruitvision LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ncnn_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.tools/ncnn/ncnn-20261020-android/${ANDROID_ABI}/lib/cmake/ncnn")
find_package(ncnn REQUIRED)
# NCNN is built without exceptions; the JNI boundary still needs local error handling.
get_target_property(ncnn_options ncnn INTERFACE_COMPILE_OPTIONS)
list(REMOVE_ITEM ncnn_options "-fno-exceptions")
set_target_properties(ncnn PROPERTIES INTERFACE_COMPILE_OPTIONS "${ncnn_options}")
add_library(fruitvision SHARED fruit_detector.cpp)
target_link_libraries(fruitvision PRIVATE ncnn android jnigraphics log)
target_link_options(fruitvision PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
target_compile_options(fruitvision PRIVATE -Wall -Wextra -Werror=return-type -fexceptions)
